### [1929. Concatenation of Array](https://leetcode.com/problems/concatenation-of-array/)

```cpp
class Solution {
private:
    static const int COPY_NUMBER = 2;

public:
    vector<int> getConcatenation(vector<int>& nums) {
        // edge case
        if (nums.empty()) return {};

        std::vector<int> res;
        res.reserve(nums.size() * COPY_NUMBER);

        for (int i = 1; i <= COPY_NUMBER; ++i) {
            res.insert(res.end(), nums.begin(), nums.end());
        }

        return res;
    }
};
```

### [217. Contains Duplicate](https://leetcode.com/problems/contains-duplicate/)

```cpp
class Solution {
public:
    bool containsDuplicate(vector<int>& nums) {
        // edge case
        if (nums.empty()) return false;

        std::unordered_set<int> uset;

        for (const auto num : nums) {
            if (uset.count(num)) return true;
            uset.insert(num);
        }

        return false;
    }
};
```

### [242. Valid Anagram](https://leetcode.com/problems/valid-anagram/)

```cpp
class Solution {
public:
    bool isAnagram(string s, string t) {
        // edge case
        if (s.size() != t.size()) return false;

        std::unordered_map<char, int> umap; // key: char, value: frequency

        for (const auto ch : s) {
            ++umap[ch];
        }

        for (const auto ch : t) {
            --umap[ch];
        }

        for (const auto it : umap) {
            if (it.second != 0) return false;
        }

        return true;
    }
};
```

```cpp
class Solution {
public:
    vector<int> twoSum(vector<int>& nums, int target) {
        // edge case
        if (nums.empty()) return {};

        std::unordered_map<int, int> umap; // key: num, value: index

        for (int i = 0; i < nums.size(); ++i) {
            if (umap.count(target - nums[i])) return {umap[target - nums[i]], i};
            umap[nums[i]] = i;
        }

        return {};
    }
};
```

### [14. Longest Common Prefix](https://leetcode.com/problems/longest-common-prefix/)

```cpp
class Solution {
public:
    string longestCommonPrefix(vector<string>& strs) {
        // edge case
        if (strs.empty()) return "";
        if (strs.size() == 1) return strs[0];

        for (int i = 0; i < strs[0].size(); ++i) {
            for (int j = 1; j < strs.size(); ++j) {
                if (i >= strs[j].size() || strs[j][i] != strs[0][i]) {
                    return strs[0].substr(0, i);
                }
            }
        }

        return strs[0];
    }
};
```
